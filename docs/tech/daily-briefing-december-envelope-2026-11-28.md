BEGIN READ ONLY;
SELECT current_setting('transaction_read_only') AS read_only,
       public.read_hosting_release_status() AS release_status,
       public.read_monthly_provider_budget() AS monthly_provider_budget;
WITH input AS MATERIALIZED (
  SELECT public.daily_briefing_input(s.reader_id, '2026-11-30 07:00Z',
                                     '2026-12-01 07:00Z', DATE '2026-12-01') AS payload
  FROM public.daily_briefing_schedules s
  JOIN public.reader_collection_policies p ON p.user_id = s.reader_id AND p.enabled
  WHERE s.enabled AND EXISTS (SELECT 1 FROM public.daily_briefing_control WHERE id AND enabled)
)
SELECT count(*) AS eligible_readers,
       sum(jsonb_array_length(payload->'sources')) AS followed_sources,
       (SELECT count(*) FROM input i CROSS JOIN LATERAL jsonb_array_elements(i.payload->'sources') s
        WHERE s->>'state' = 'checked') AS fully_checked_sources,
       sum(jsonb_array_length(payload->'updates')) AS eligible_updates,
       (SELECT coalesce(sum(jsonb_array_length(u->'findings')), 0)
        FROM input i CROSS JOIN LATERAL jsonb_array_elements(i.payload->'updates') u) AS approved_findings
FROM input;
ROLLBACK;
